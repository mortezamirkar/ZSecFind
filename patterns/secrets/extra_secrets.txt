# Additional cloud/auth secret patterns (complements core + curated sets).
# Skipped here when already covered: AIza, AWS AKIA, Stripe, Twilio SK, GitHub ghp, Slack xoxb, PEM blocks, JWT category.

# Firebase Cloud Messaging server key
|AAAA[A-Za-z0-9_-]{7}:[A-Za-z0-9_-]{140}

# Google reCAPTCHA site key
|\b6L[0-9A-Za-z-_]{38}\b
|\b6[0-9a-zA-Z_-]{39}\b

# Google OAuth access token
|\bya29\.[0-9A-Za-z\-_]+\b

# Amazon MWS auth token
|\bamzn\.mws\.[0-9a-f]{8}-[0-9a-f]{4}-[0-9a-f]{4}-[0-9a-f]{4}-[0-9a-f]{12}\b

# Amazon S3 bucket URLs
|\b[a-zA-Z0-9._-]+\.s3\.amazonaws\.com\b
|\bs3://[a-zA-Z0-9._-]+\b
|\bs3-[a-zA-Z0-9._/-]+\b
|\bs3\.amazonaws\.com/[a-zA-Z0-9._-]+\b
|\bs3\.console\.aws\.amazon\.com/s3/buckets/[a-zA-Z0-9._-]+\b

# Facebook access token
|\bEAACEdEose0cBA[0-9A-Za-z]+\b

# Authorization headers
i|\bbasic [a-zA-Z0-9=:_+\/-]{5,100}\b
i|\bbearer [a-zA-Z0-9_\-.=:_+\/]{5,100}\b
i|\bapi(?:key|_key|\s+)[a-zA-Z0-9_\-]{5,100}\b

# Mailgun API key
|\bkey-[0-9a-zA-Z]{32}\b

# Twilio Account SID / App SID
|\bAC[a-zA-Z0-9_\-]{32}\b
|\bAP[a-zA-Z0-9_\-]{32}\b

# PayPal Braintree access token
|\baccess_token\$production\$[0-9a-z]{16}\$[0-9a-f]{32}\b

# Square (extra variants beyond curated sq0atp/sq0csp)
|\bsq0[a-z]{3}-[0-9A-Za-z\-_]{22,43}\b
|\bsqOatp-[0-9A-Za-z\-_]{22}\b
|\bEAAA[a-zA-Z0-9]{60}\b

# GitHub credentials embedded in URL (user:token@github.com)
|\b[a-zA-Z0-9_-]+:[a-zA-Z0-9_\-]+@github\.com\b

# PEM / PGP key headers (BEGIN-only; complements full PEM block in curated)
|-----BEGIN RSA PRIVATE KEY-----
|-----BEGIN DSA PRIVATE KEY-----
|-----BEGIN EC PRIVATE KEY-----
|-----BEGIN PGP PRIVATE KEY BLOCK-----

# Slack api_token JSON field
|"api_token"\s*:\s*"(xox[a-zA-Z]-[a-zA-Z0-9-]+)"

# Inline password / pwd / passwd assignments
i|(?:password\s*[`=:"]+\s*[^\s]+|password is\s*[`=:"]*\s*[^\s]+|pwd\s*[`=:"]*\s*[^\s]+|passwd\s*[`=:"]+\s*[^\s]+)
